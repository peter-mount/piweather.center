\sectionpaged{Feels Like}
\index{Feels Like}
\index{Temperature!Feels Like}

Feels Like is the temperature a human perceives based on the actual temperature, relative humidity and wind speed.

\NewValueUnit{\FeelsLike}{T_{fl}}{Feels Like}
\begin{definition}Calculating Feels Like:
\begin{eqnarray}
    \FeelsLike &=&
    \begin{cases}
        \WindChill = \textbf{WindChill}( \AmbientTemperature, \WindSpeed ) & \quad \textbf{if } \AmbientTemperature \leq \Fahrenheit{50}\,and\,\WindSpeed \geq 3 mph\\
        \HeatIndex = \textbf{HeatIndex}( \AmbientTemperature, \RelativeHumidity) & \quad \textbf{if } \AmbientTemperature \geq \Fahrenheit{80} \\
        \AmbientTemperature & \quad \textbf{otherwise}
    \end{cases}
\end{eqnarray}

\WhereDescription{
    \item[$\FeelsLike$]             The Feels like temperature in degrees \Fahrenheit{}
    \item[$\AmbientTemperature$]    the air temperature in degrees \Fahrenheit{}
    \item[$\WindSpeed$]             the wind speed in miles per hour
    \item[$\RelativeHumidity$]      Relative Humidity
}
\end{definition}
